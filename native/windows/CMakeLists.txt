cmake_minimum_required(VERSION 3.21)
project(Salpakan LANGUAGES CXX RC)
get_filename_component(GAME_HTML "${CMAKE_CURRENT_SOURCE_DIR}/../../dist/Salpakan.html" ABSOLUTE)
if(NOT EXISTS "${GAME_HTML}")
  message(FATAL_ERROR "Run python tools/build.py from the repository first.")
endif()
configure_file(game.rc.in "${CMAKE_CURRENT_BINARY_DIR}/game.rc" @ONLY)
set_property(SOURCE "${CMAKE_CURRENT_BINARY_DIR}/game.rc" PROPERTY OBJECT_DEPENDS "${GAME_HTML};${CMAKE_CURRENT_SOURCE_DIR}/../../tests/native/play.js")
add_executable(Salpakan WIN32 main.cpp "${CMAKE_CURRENT_BINARY_DIR}/game.rc")
target_compile_features(Salpakan PRIVATE cxx_std_17)
target_compile_options(Salpakan PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/utf-8>")
target_compile_definitions(Salpakan PRIVATE UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
set_property(TARGET Salpakan PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_include_directories(Salpakan PRIVATE "${WEBVIEW2_SDK}/build/native/include")
target_link_libraries(Salpakan PRIVATE "${WEBVIEW2_SDK}/build/native/x64/WebView2LoaderStatic.lib"
    ole32 oleaut32 shell32 shlwapi user32 version advapi32)
